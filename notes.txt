//write the Dockerfile step by step

// pulling an image
docker pull mongo:version

// make sure to write the dependency image pull i.e FROM node:22
// then build the docker image , ( . is for src file location )
docker build -t node-app:1.0 .

// to make a container from images 
// -d == detach so that we can use the terminal further
// -p host: container  == configures which host port -> container port
// -e == configures environment variables for that container
docker run -d \
 -p 3000:3000 \
 -e MONGO_DB_USERNAME=admin -e MONGO_DB_PASSWORD=pass \
 --net my_network \
 --name containerName \
 node-app:1.0 

// container already there but not running
docker stop/start/pause containerId 

//view docker networks
docker network ls 

// creating a network for communication inside container
docker network create my_network

// view running container  
docker ps  // -a too see all containers running/not-running instead of going to history

// view images present locally
docker images

// access logs of a container
docker logs containerId

// access the cmd of the container
docker exec -it containerId /bin/bash(or sh) //it = interactive shell
